#include <iostream>
#include <string>
using namespace std;

class Employee {
private:
    int empId;
    string name;
    double salary;
public:
    // Default Constructor
    Employee() {
        empId = 0;
        name = "Not Assigned";
        salary = 0.0;
    }

    // Parameterized Constructor
    Employee(int id, string empName, double sal) {
        empId = id;
        name = empName;
        salary = sal;
    }

    void calculateTotalSalary(double bonus) {
        double totalSalary = salary + bonus;
        cout << "\n--- Employee Details ---" << endl;
        cout << "Employee ID: " << empId << endl;
        cout << "Name: " << name << endl;
        cout << "Base Salary: " << salary << endl;
        cout << "Bonus: " << bonus << endl;
        cout << "Total Salary: " << totalSalary << endl;
    }
};

int main() {
    Employee emp1(101, "Alice Smith", 50000.0);
    emp1.calculateTotalSalary(5000.0);

    Employee emp2;
    cout << "\nUsing Default Constructor Object:" << endl;
    emp2.calculateTotalSalary(1000.0);

    return 0;
}
