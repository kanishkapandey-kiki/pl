#include <iostream>
#include <string>
using namespace std;

class Book {
private:
    string title;
    string author;
    float price;
public:
    // Parameterized Constructor
    Book(string t, string a, float p) {
        title = t;
        author = a;
        price = p;
        cout << "Parameterized Constructor Called for: " << title << endl;
    }

    // Copy Constructor
    Book(const Book &b) {
        title = b.title;
        author = b.author;
        price = b.price;
        cout << "Copy Constructor Called for: " << title << endl;
    }

    void display() const {
        cout << "Title: " << title << ", Author: " << author << ", Price: " << price << endl;
    }

    // Destructor
    ~Book() {
        cout << "Destructor Called for: " << title << endl;
    }
};

int main() {
    cout << "Creating original book object:" << endl;
    Book book1("C++ Programming", "Bjarne Stroustrup", 450.50);

    cout << "\nCreating copied book object:" << endl;
    Book book2(book1);

    cout << "\nDisplaying Original Book:" << endl;
    book1.display();

    cout << "\nDisplaying Copied Book:" << endl;
    book2.display();

    cout << "\nEnd of main function:" << endl;
    return 0;
}
