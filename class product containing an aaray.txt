#include <iostream>
#include <string>
using namespace std;

class Product {
private:
    int productId;
    string productName;
    float price;
    float monthlySales[12];
public:
    void accept() {
        cout << "Enter Product ID: ";
        cin >> productId;
        cout << "Enter Product Name: ";
        cin.ignore();
        getline(cin, productName);
        cout << "Enter Price per unit: ";
        cin >> price;
        cout << "Enter monthly sales for 12 months:\n";
        for (int i = 0; i < 12; i++) {
            cout << "Month " << i + 1 << ": ";
            cin >> monthlySales[i];
        }
    }

    void display() const {
        float totalQuantity = 0;
        for (int i = 0; i < 12; i++) {
            totalQuantity += monthlySales[i];
        }
        float totalBill = totalQuantity * price;

        cout << "\n--- Product Details ---" << endl;
        cout << "Product ID: " << productId << endl;
        cout << "Product Name: " << productName << endl;
        cout << "Price per unit: " << price << endl;
        cout << "Total Quantity Sold (12 months): " << totalQuantity << endl;
        cout << "Total Bill / Revenue: " << totalBill << endl;
    }
};

int main() {
    int n;
    cout << "Enter number of products: ";
    cin >> n;

    Product products[10]; // Adjustable array size
    for (int i = 0; i < n; i++) {
        cout << "\nEnter details for product " << i + 1 << ":" << endl;
        products[i].accept();
    }

    for (int i = 0; i < n; i++) {
        products[i].display();
    }

    return 0;
}
